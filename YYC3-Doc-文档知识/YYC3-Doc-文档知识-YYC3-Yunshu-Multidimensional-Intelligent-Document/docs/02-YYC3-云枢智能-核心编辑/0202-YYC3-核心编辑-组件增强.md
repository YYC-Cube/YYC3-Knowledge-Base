## 二十六、日期与时间组件

## 26.1 日期组件功能

文档支持插入动态日期组件，可显示当前日期或指定日期：

```javascript
// 插入日期组件
sdk.insertDateComponent({
container: '#editor-container',
type: 'static', // static/dynamic
date: '2025-07-18', // 静态日期
format: 'yyyy年MM月dd日 星期E', // 显示格式
style: {
fontSize: '14px',
color: '#4E5969',
fontWeight: 'normal'
}
});
// 插入动态日期（自动更新）
sdk.insertDateComponent({
type: 'dynamic',
format: 'yyyy-MM-dd HH:mm:ss',
updateInterval: 1000, // 每秒更新
style: {
color: '#1D2129'
}
});

```

## 26.2 时间戳与倒计时集成

```javascript

// 插入时间戳
sdk.insertTimestamp({
container: '#editor-container',
timestamp: Date.now(),

format: 'relative', // 相对时间（如"3小时前"）
refresh: true // 是否自动刷新
});
// 倒计时组件扩展
sdk.insertCountdown({
target: '2025-12-31T23:59:59',
format: 'D天H时M分S秒',
expiredText: '已过期',
style: {
digitColor: '#F53F3F',
separatorColor: '#4E5969'
}
});

```

## 二十八、AI 辅助功能实现

28.1 AI 帮我写功能
文档集成 AI 辅助写作功能，可自动生成内容：

```javascript
// 调用AI帮我写
sdk.ai.write({
prompt: '写一段关于产品迭代的会议纪要开头',
style: 'formal', // 正式/简洁/详细
length: 'medium', // 短/中/长
onProgress: (partialResult) => {
// 实时显示生成中的内容
updateEditorContent(partialResult);
}
}).then(fullResult => {
console.log('AI生成完成:', fullResult);
});
// 基于选中内容扩展
sdk.ai.expandSelection({
direction: 'detail', // 详细展开/举例说明/总结
count: 3 // 生成3个要点
}).then(expandedContent => {
sdk.insertAfterSelection(expandedContent);
});

```

## 28.2 AI 帮我改功能

AI 可优化已有内容，调整风格或修正错误：

```javascript
// 调用AI帮我改
sdk.ai.revise({
content: '这个产品很好，大家都喜欢用',
action: 'polish', // 润色/简化/专业术语转换/翻译
targetLanguage: 'en' // 翻译目标语言
}).then(revisedContent => {
console.log('修改后:', revisedContent);
// 替换选中内容
sdk.replaceSelection(revisedContent);
});

// 语法检查与纠错
sdk.ai.checkGrammar({
content: '我们已经完成了各项任务，明天将进行了测试'
}).then(result => {
if (result.hasErrors) {
showGrammarErrors(result.errors);
// 自动修正
sdk.replaceSelection(result.correctedContent);
}
});

```

## 二十九、内容保护区与权限控制

## 29.1 内容保护区设置

文档支持设置内容保护区，仅限指定用户编辑：

```javascript
// 创建内容保护区
sdk.createProtectedRegion({

end: 300, // 结束位置

}
});
// 检查当前用户是否有权限编辑
sdk.canEditRegion(regionId).then(canEdit => {
if (!canEdit) {
showTooltip('无编辑权限，此区域受保护');
}
});

```

## 三十、扩展插件系统

## 30.1 插件注册与管理

文档支持通过插件扩展功能，如添加特殊格式或第三方服务集成：

```javascript
// 注册插件
sdk.registerPlugin({
id: 'weather-plugin',
name: '天气插件',
version: '1.0.0',

icon: '🌤️',

activate: (sdk) => {
// 注册工具栏按钮
sdk.addToolbarButton({
id: 'weather-button',

icon: '🌤️',

onClick: () => insertWeatherWidget(sdk)
});
},
deactivate: () => {
// 清理资源
sdk.removeToolbarButton('weather-button');
}
});
// 插入天气组件

function insertWeatherWidget(sdk) {
sdk.insertComponent({
type: 'weather',
data: {
city: '北京',
unit: 'celsius'
},
renderer: (data) => `
<div class="weather-widget">
<div class="city">${data.city}</div>
<div class="loading">加载中...</div>
</div>
`,
loader: async (element, data) => {
// 加载天气数据
const weather = await fetchWeather(data.city);
element.innerHTML = `
<div class="city">${data.city}</div>
<div class="temp">${weather.temp}°${data.unit === 'celsius' ? 'C' : 'F'}</div>
<div class="desc">${weather.desc}</div>
`;
}
});
}

```

## 三十一、分栏布局功能实现

文档支持分栏布局，可灵活设置栏数与间距，实现专业排版效果。核心实现代码如下：

```javascript
// 插入分栏布局
sdk.insertColumnLayout({
container: '#editor-container',
columns: 2, // 分栏数量（1-4）
gutter: 24, // 栏间距（px）
width: '80%', // 总宽度
alignment: 'center', // 对齐方式
content: [
'<p>第一栏内容...这是一段分栏排版的示例文本，用于展示多栏布局效果。</p>',
'<p>第二栏内容...分栏功能适合长文档阅读，可优化页面空间利用率。</p>'
]
});
// 动态调整分栏
sdk.updateColumnLayout(columnId, {
columns: 3,
gutter: 16,
responsive: true // 启用响应式，小屏幕自动合并为单栏
});

分栏样式定义：

css
.column-container {
display: flex;
width: 100%;
margin: 16px 0;
}

.column-item {
flex: 1;
padding: 0 8px;

}

.column-gutter {
width: 24px;
flex-shrink: 0;
}
/* 响应式调整 */
@media (max-width: 768px) {
.column-container.responsive {
flex-direction: column;
}
.column-gutter {
height: 16px;
width: 100%;
}
}

```
