## 256.1 动态摘要的智能调整
支持根据文档内容实时变化自动更新摘要：

```javascript
// 配置摘要实时更新机制
sdk.configureSummaryAutoUpdate({
triggerOn: ['content-add', 'content-remove', 'structure-change'],
updateDelay: 5000, // 内容变化后延迟5秒更新
preserveKeyPoints: true, // 保留核心要点
adjustLength: true // 自动调整长度
});
// 手动触发摘要优化
sdk.optimizeSummary({
focus: 'recent-changes', // 聚焦最新变更
length: 'auto',
includeCitations: true
}).then(updatedSummary => {
sdk.replaceSummary(updatedSummary, { showDiff: true });
});

```

## 265.1 工作流自定义节点开发
支持创建自定义工作流节点扩展流程能力：

```javascript
// 注册自定义工作流节点
sdk.registerCustomWorkflowNode({
nodeId: 'content-validation',
name: '内容验证',
description: '自动检查文档内容完整性',
parameters: [
{ name: 'checkSpelling', type: 'boolean', default: true },
{ name: 'minWordCount', type: 'number', default: 500 }
],
implementation: async (params, context) => {
const result = await validateDocumentContent(context.docId, params);
return { status: result.valid ? 'passed' : 'failed', details: result.issues };
}
});
// 在工作流中使用自定义节点
sdk.addCustomNodeToWorkflow('content-approval', {
nodeId: 'content-validation',
position: 'after-submission', // 插入位置
parameters: { minWordCount: 1000 }
});

```
