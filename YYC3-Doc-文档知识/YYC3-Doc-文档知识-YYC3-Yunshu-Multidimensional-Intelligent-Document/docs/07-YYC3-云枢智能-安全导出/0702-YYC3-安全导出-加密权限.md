## 196.1 分级加密策略
支持按内容敏感度设置不同加密级别：

```javascript
// 配置分级加密
sdk.configureTieredEncryption({
levels: [
{
id: 'public',
encryption: false,
access: 'anyone'
},
{
id: 'internal',
encryption: true,
algorithm: 'AES-128',
access: 'organization'
},
{
id: 'confidential',
encryption: true,
algorithm: 'AES-256',
access: 'restricted',
watermark: '机密'
}
],
defaultLevel: 'internal'
});
// 为文档片段设置加密级别
sdk.setContentEncryptionLevel({
range: { start: 2000, end: 5000 },
level: 'confidential'
});

```

```javascript

// 配置基于时间的动态权限
sdk.setTimeBasedPermissions({
```

## 197.1 多表格数据联动
支持不同 sheet 间的单元格引用与计算：

```javascript
// 跨sheet引用数据
sdk.table.setCrossSheetReference({
source: { sheetId: 'sales-data', range: 'B2:B10' },
target: { sheetId: 'summary', cell: 'C3' },
type: 'dynamic', // 动态/静态引用
refreshOnChange: true
});
// 跨sheet计算公式
sdk.table.setCrossSheetFormula({
sheetId: 'summary',
cell: 'D5',
formula: 'SUM(sales-data!B2:B10) * 0.15'
});

```

## 198.1 基于内容的图片替换
支持根据图片内容特征查找替换选项：

```javascript
// 智能图片替换建议
sdk.suggestImageReplacements(imageId, {
criteria: ['subject', 'style', 'quality'], // 匹配标准
count: 5, // 推荐5张替换图片
source: 'library' // 从图库/网络/上传中推荐
}).then(suggestions => {
sdk.displayImageReplaceOptions(suggestions.map(s => ({
thumbnail: s.preview,
similarity: s.score,
replace: () => sdk.replaceImage(imageId, s.url)
})));
});
// 批量优化图片质量
sdk.batchEnhanceImages({
imageIds: ['img-801', 'img-802'],
actions: ['enhance', 'compress', 'resize'],
targetQuality: 0.85
});

```

```javascript
});
```

```javascript
});
```
