## 226.1 标签动态更新机制
支持基于文档内容变化自动调整标签体系：

```javascript
// 配置标签自动演化规则
sdk.configureTagEvolution({
autoUpdate: true,
sensitivity: 0.6, // 内容变化敏感度
mergeThreshold: 0.7, // 标签合并阈值
pruneUnused: 30 // 30天未使用标签自动移除
});
// 触发标签重新计算
sdk.evolveDocumentTags({
forceRecalculation: true,
includeRelatedDocs: false // 不影响关联文档
}).then(updatedTags => {
sdk.updateTagDisplay(updatedTags, {
showChanges: true // 高亮新增/移除标签
});
});

```

## 234.1 文档内容字幕化
支持将多语言文档内容转换为同步字幕格式：

```javascript
// 生成文档内容字幕
sdk.generateDocumentSubtitles({
langs: ['zh-CN', 'en-US', 'es-ES'],
segmentBy: 'sentence', // 按句子/段落分割
timing: 'reading-speed', // 基于阅读速度/固定间隔
format: 'srt' // srt/vtt/ass格式
}).then(subtitles => {
Object.entries(subtitles).forEach(([lang, data]) => {
sdk.attachSubtitleFile(data, { lang, name: `doc-sub-${lang}` });
});
});
// 字幕与内容同步
sdk.syncSubtitlesWithContent({
lang: 'en-US',
highlight: true, // 播放时高亮对应文本
offset: 500 // 字幕偏移量(ms)
});

```
