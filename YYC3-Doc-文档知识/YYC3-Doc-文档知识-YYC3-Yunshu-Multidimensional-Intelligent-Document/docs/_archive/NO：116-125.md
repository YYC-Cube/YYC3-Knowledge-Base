一百一十六、拼写检查与自动更正

## 116.1 实时拼写检查
支持多语言实时拼写错误检测：

```javascript
// 启用拼写检查
sdk.enableSpellCheck({
languages: ['zh-CN', 'en-US'], // 支持语言
mode: 'real-time', // 实时/手动
dictionary: {
custom: ['云枢', '多维文档', 'YYC³'] // 自定义词典
},
ignore: ['technical-terms', 'acronyms'] // 忽略技术术语和缩写
});
// 拼写错误处理
sdk.on('spelling-error', (error) => {
// 标记错误
sdk.markTextRange(error.start, error.end, {
type: 'spelling',
tooltip: `建议: ${error.suggestions.join(', ')}`
});
});

```

## 116.2 自动更正规则

```javascript

// 配置自动更正规则
sdk.configureAutoCorrect({
replacements: [
{ from: 'teh', to: 'the' },
{ from: '云枢 ', to: 'YYC³ ' },
{ from: '//', to: '// ' } // 自动添加空格
],
smartQuotes: true, // 自动替换引号为智能引号

fractionCorrection: true // 自动替换1/2为½等
});
// 临时禁用自动更正
sdk.toggleAutoCorrect(false);
```

## 117.1 复杂合并操作
支持不规则单元格合并：

```javascript

sdk.table.mergeCells(tableId, {
startRow: 1,
startCol: 1,
endRow: 3,
endCol: 2,
alignment: 'center' // 合并后内容对齐方式
});
// 按条件合并
sdk.table.mergeCellsByValue(tableId, {
column: 0, // 按第一列值合并
ignoreEmpty: true // 忽略空值
});

```

```javascript

cols: 2 // 拆分为2列
});
// 取消所有合并
sdk.table.unmergeAllCells(tableId);
```

```javascript
autoResolve: true // 自动解决简单冲突
});
// 监听冲突事件
sdk.on('shortcut-conflict', (conflict) => {
showConflictNotification({
commandA: conflict.command1,

commandB: conflict.command2,
shortcut: conflict.shortcut,
resolution: conflict.autoResolved ? '已自动解决' : '需手动处理'
});
});

```
