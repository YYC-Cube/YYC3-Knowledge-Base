## 146.1 多文档批量格式调整
支持同时对多个文档执行统一格式修改：

```javascript
// 批量设置文档格式
sdk.batchSetDocumentFormat({
docIds: ['doc-001', 'doc-002', 'doc-003'],
format: {
fontFamily: 'Inter, sans-serif',
fontSize: 14,
lineHeight: 1.6,
headingStyles: {
level1: { color: '#1890FF', fontSize: 20 },
level2: { color: '#4E5969', fontSize: 18 }
}
},
onProgress: (docId, progress) => {
console.log(`文档${docId}格式调整进度：${progress}%`);
}
}).then(results => {
console.log(`成功调整${results.successCount}个文档格式`);
});

```

## 146.2 自动化格式校验

```javascript

// 批量校验文档格式合规性
sdk.batchValidateDocumentFormat({
docIds: ['doc-004', 'doc-005'],

maxHeadingLevel: 3,
requiredStyles: { heading1: { color: '#1890FF' } },
forbiddenFonts: ['Comic Sans MS']
}

}).then(reports => {
reports.forEach(report => {
if (!report.valid) {
console.log(`文档${report.docId}格式问题：`, report.issues);
}
});
});
```

```javascript

relatednessThreshold: 0.6, // 关联度阈值
limit: 10,
includeContentSnippets: true,
filter: {
createdAfter: '2025-01-01',
author: ['dev-team']
}
}).then(results => {
renderSearchResults({
query: '用户登录流程',
results: results.map(r => ({
docId: r.docId,
title: r.title,
relevance: r.score,
snippet: r.snippet
}))
});
});

```

## 151.1 多维度摘要提取
支持按不同维度生成文档摘要：

```javascript
// 生成多维度文档摘要
sdk.ai.generateMultiDimensionSummary({
dimensions: ['keyPoints', 'structure', 'arguments', 'conclusions'],
detailLevel: 'comprehensive', // 简要/详细/全面
format: 'markdown',
lengthRatio: 0.2 // 摘要长度为原文20%
}).then(summaries => {
sdk.insertContent(`# 文档多维度摘要\n
## 核心要点\n${summaries.keyPoints}\n
## 结构分析\n${summaries.structure}`, { position: 'top' });
});
// 生成章节级摘要
sdk.ai.generateChapterSummaries({
targetLevels: [1, 2], // 基于H1和H2生成
mergeSimilar: true // 合并相似章节摘要
});

```

```javascript

docIds: ['doc-101', 'doc-102', 'doc-103'],
format: 'pdf',
options: {
includeHeadersFooters: true,
watermark: '内部资料',

pageRange: 'all'
```
