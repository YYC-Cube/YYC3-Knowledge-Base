一百三十六、文档模板变量的跨文档引用

## 136.1 外部变量引入
支持引用其他文档中的模板变量值：

```javascript
// 配置跨文档变量引用
sdk.configureCrossDocumentVariables({
sourceDocId: 'product-spec',
variables: ['version', 'releaseDate'],
refreshInterval: 3600000 // 1小时刷新一次
});
// 在当前模板中使用外部变量
sdk.registerContentTemplate({
id: 'release-notes',
variables: [
{
name: 'productVersion',
type: 'cross-reference',
source: { docId: 'product-spec', variable: 'version' }
}
],
content: `# 版本说明 v{{productVersion}}
发布日期：{{releaseDate}}
`
});

```

```javascript

// 对图片执行OCR

languages: ['zh-CN', 'en-US'],
detectOrientation: true, // 自动检测方向
outputFormat: 'text' // text/html/json

onComplete: (imageId, result) => {

}
});

```

## 142.1 离线模式配置
支持无网络环境下的文档编辑与后续同步：

```javascript
// 启用离线编辑
sdk.enableOfflineMode({
autoSync: true, // 恢复网络后自动同步
storageLimit: 500 * 1024 * 1024, // 500MB存储上限
priority: ['unsaved', 'recent'] // 优先保存未保存和最近编辑文档
});
// 监听离线状态变化
sdk.on('offline-status-change', (isOffline) => {
updateStatusIndicator(isOffline ? '离线模式' : '在线同步中');
if (isOffline) {
showOfflineNotification('已切换至离线编辑，更改将在联网后同步');
}
});

```
