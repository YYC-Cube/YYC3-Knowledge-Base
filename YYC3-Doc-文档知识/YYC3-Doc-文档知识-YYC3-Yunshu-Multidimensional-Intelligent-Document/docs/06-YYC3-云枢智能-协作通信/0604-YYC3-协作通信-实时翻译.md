## 156.1 协作会话实时翻译
支持多人协作时的实时语言转换：

```javascript
// 启用协作实时翻译
```

## 157.2 自定义函数注册

```javascript

// 注册表格自定义函数
sdk.table.registerCustomFunction({
name: 'RANK_PERCENTILE',
description: '计算数值在数据集中的百分位排名',
parameters: ['value', 'range'],
implementation: (value, range) => {
const sorted = [...range].sort((a, b) => a - b);
const rank = sorted.findIndex(v => v >= value) + 1;
return ((rank / sorted.length) * 100).toFixed(2);
}
});
```
